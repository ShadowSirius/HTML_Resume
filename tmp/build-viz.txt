  buildViz() {
    const h = React.createElement, W = 900, H = 1260, cx = 450, groundY = 860;
    const printing = !!this.state.printing, sel = printing ? null : this.state.selected;
    const font = 'Arial, sans-serif', fs = 15.5, lineH = 18.5;
    const measure = this._measure || (this._measure = document.createElement('canvas').getContext('2d'));
    const widthOf = (s, bold = false) => { measure.font = (bold ? '700 ' : '400 ') + fs + 'px Arial'; return measure.measureText(s).width; };
    const wrap = (s, width, bold = false) => {
      const lines = []; let line = '';
      String(s).split(/\s+/).forEach(word => {
        if (line && widthOf(line + ' ' + word, bold) > width) { lines.push(line); line = ''; }
        line += (line ? ' ' : '') + word;
      });
      if (line) lines.push(line);
      return lines;
    };
    const bNodes = [], bLinks = [], TREE = this.visibleTree();
    // ponytail: three compact columns fit the current inventory; additions need a print-fit check.
    const colY = [groundY + 34, groundY + 34, groundY + 34];
    const add = (data, x, y, parent, cat, kind, lines) => {
      // Layout metadata belongs to copies: parent links must never pollute exported skill data.
      const n = { ...data, px: x, py: y, parent, cat, kind, lines, gen: parent ? parent.gen + 1 : 1 };
      bNodes.push(n); bLinks.push(n); return n;
    };
    const place = (parent, left, right, start) => {
      const x0 = left + 16;
      let x = x0, y = start;
      const flow = (data, p, rib) => {
        const group = !!data.children?.length;
        const lines = wrap(data.name + (group ? ':' : ''), right - x0 - 9, group);
        const width = Math.max(...lines.map(s => widthOf(s, group))) + 15;
        if (x > x0 && (x + width > right || lines.length > 1)) { x = x0; y += lineH; }
        const n = add(data, x, y, p, parent.cat, group ? 'group' : 'terminal', lines);
        n.rib = rib;
        x += width;
        if (lines.length > 1) { x = x0; y += lines.length * lineH; }
        (data.children || []).forEach(child => flow(child, n, false));
      };
      (parent.children || []).forEach(data => {
        if (x > x0) { x = x0; y += lineH; }
        flow(data, parent, true);
      });
      return y + (x > x0 ? lineH : 0);
    };
    (TREE.children || []).forEach((cat, i) => {
      const col = colY.indexOf(Math.min(...colY)), left = 20 + col * 294, right = left + 282, top = colY[col];
      const lines = wrap(cat.name, right - left - 22, true);
      const n = add(cat, left + 6, top, null, cat.name, 'root', lines);
      n.spineX = left - 2;
      colY[col] = place(n, left, right, top + lines.length * lineH + 6) + 14;
    });
    const drag = this.state.drag || {};
    bNodes.forEach(n => {
      const own = drag['n:' + n.id] || { dx: 0, dy: 0 }, pc = n.parent?._cum || { dx: 0, dy: 0 };
      n._cum = { dx: pc.dx + own.dx, dy: pc.dy + own.dy };
      n.px += n._cum.dx; n.py += n._cum.dy;
    });
    const actSet = PROJECTS[sel] ? new Set(PROJECTS[sel].activeSkills || []) : null;
    const hiSet = new Set(), ancSet = new Set();
    if (actSet) bNodes.forEach(n => { if (actSet.has(n.id)) { hiSet.add(n); let p = n.parent; while (p) { ancSet.add(p); p = p.parent; } } });
    const dimB = n => actSet && !hiSet.has(n) && !ancSet.has(n);
    let order = Object.keys(PROJECTS).sort((a, b) => parseInt(PROJECTS[a].year) - parseInt(PROJECTS[b].year));
    if ((this.props.sortDirection || 'oldest-bottom') === 'newest-bottom') order.reverse();
    const cardW = 352, cardH = 140, rowGap = Math.min(148, 740 / Math.max(1, Math.ceil(order.length / 2))), cardScale = Math.min(1, rowGap / 148);
    const briefs = order.map((id, k) => {
      const left = k % 2 === 0, row = Math.ceil(order.length / 2) - 1 - Math.floor(k / 2);
      const dc = drag['c:' + id] || { dx: 0, dy: 0 }, x = (left ? 22 : 526) + dc.dx, y = 52 + row * rowGap + (left ? 0 : 48 * cardScale) + dc.dy;
      return { id, left, row, x, y, tipX: x + cardW / 2, tipY: y + cardH / 2 };
    });
    const els = [], ink = '#71877e', accent = '#426a61';
    const handles = [], snap = v => Math.round(v / 10) * 10;
    const handle = (key, x, y, move, label) => {
      if (printing || !this.state.routeEdit) return;
      handles.push(h('circle', { key:key+label, className: 'route-handle', 'data-route-key': key, cx:x, cy:y, r:6, fill:'#fff', stroke:accent, strokeWidth:2, tabIndex:0, role:'slider', 'aria-label':label, 'aria-valuetext':Math.round(x)+', '+Math.round(y),
        onPointerDown:e => this.routeStart(e,key,move), onClick:e => e.stopPropagation(),
        onKeyDown:e => { const delta = {ArrowLeft:[-10,0],ArrowRight:[10,0],ArrowUp:[0,-10],ArrowDown:[0,10]}[e.key]; if (!delta) return; e.preventDefault(); const value = move({x:x+delta[0],y:y+delta[1]}); this.setState(s => ({drag:{...s.drag,[key]:value}}), () => { try {localStorage.setItem('resumeTreeDragV1',JSON.stringify(this.state.drag));}catch(e){} }); }, style:{cursor:'move',touchAction:'none'} }));
    };
    const baseT = { fontFamily: font, paintOrder: 'stroke', stroke: '#fff', strokeLinejoin: 'round' };
    els.push(h('defs', {}, h('filter', { id: 'nodeGlow', x: '-100%', y: '-100%', width: '300%', height: '300%' }, h('feGaussianBlur', { stdDeviation: 2.2, result: 'blur' }), h('feMerge', {}, h('feMergeNode', { in: 'blur' }), h('feMergeNode', { in: 'SourceGraphic' })))));
    els.push(h('line', { className: 'ground-line', key: 'ground', x1: 24, x2: 876, y1: groundY, y2: groundY, stroke: ink, strokeWidth: 1.1 }));
    els.push(h('text', { key: 'section', x: 24, y: 16, fontFamily: font, fontSize: 12, fontWeight: 700, letterSpacing: 1.5, fill: accent }, 'PROJECT CANOPY'));
    els.push(h('text', { key: 'roots', x: 876, y: groundY - 12, textAnchor: 'end', fontFamily: font, fontSize: 12, fontWeight: 700, letterSpacing: 1.5, fill: accent }, 'SKILL ROOTS'));
    const gStyle = this.props.rootStyle || 'straight', lStyle = this.state.linkStyle || {};
    bLinks.forEach((n, i) => {
      const p = n.parent, sx = p ? p.px : cx, sy = p ? p.py + 6 : groundY, style = lStyle[n.id] || gStyle;
      let d;
      if (!p) {
        const route = drag['r:' + n.id], spine = route?.x ?? n.spineX + n._cum.dx, busY = groundY + 13;
        if (style === 'curved') d = `M${cx},${groundY} Q${cx},${busY} ${cx - 10},${busY} L${spine + 7},${busY} Q${spine},${busY} ${spine},${busY + 7} L${spine},${n.py - 7} Q${spine},${n.py} ${n.px},${n.py}`;
        else if (style === 'stepped') d = `M${cx},${groundY} L${cx},${busY} L${spine},${busY} L${spine},${n.py} L${n.px},${n.py}`;
        else {
          const top = route?.y ?? groundY+18, bottom = Math.max(top,Math.min(n.py-4,route?.bottom ?? n.py-Math.abs(n.px-spine)));
          d = `M${cx},${groundY}L${spine},${top}V${bottom}L${n.px},${n.py}`;
          const move = p => ({...route,x:Math.max(10,Math.min(n.px-6,cx+snap(p.x-cx))),y:Math.max(groundY+10,Math.min(bottom,snap(p.y)))});
          handle('r:'+n.id,spine,top,move,n.name+' upper junction');
          handle('r:'+n.id,spine,bottom,p => ({...route,x:Math.max(10,Math.min(n.px-6,cx+snap(p.x-cx))),y:top,bottom:Math.max(top,Math.min(n.py-4,snap(p.y)))}),n.name+' lower junction');
        }
      }
      else if (n.rib) {
        const ribY = n.py - 8;
        if (style === 'curved') d = `M${sx},${sy} L${sx},${ribY - 6} Q${sx},${ribY} ${sx + 6},${ribY} L${n.px - 9},${ribY} Q${n.px},${ribY} ${n.px},${n.py}`;
        else if (style === 'stepped') d = `M${sx},${sy} L${sx},${n.py} L${n.px},${n.py}`;
        else { const key='rib:'+n.id, ry=drag[key]?.y ?? n.py-Math.abs(n.px-sx); d = `M${sx},${sy}V${ry}L${n.px},${n.py}`; handle(key,sx,ry,p=>({y:Math.max(sy,Math.min(n.py,snap(p.y)))}),n.name+' junction'); }
      }
      else {
        // Inline details share a list; their short junctions stay clear of neighboring text.
        d = style === 'curved' ? `M${n.px - 4},${n.py - 3}Q${n.px},${n.py - 3} ${n.px},${n.py}`
          : style === 'stepped' ? `M${n.px - 4},${n.py - 3}H${n.px}V${n.py}` : `M${n.px - 4},${n.py - 3}L${n.px},${n.py}`;
      }
      els.push(h('path', { key: 'rl' + i, className: 'root-link' + (n.kind === 'root' ? ' category-root-link' : ''), d, stroke: accent, strokeWidth: n.kind === 'root' ? 1.6 : 1, opacity: dimB(n) ? 0.12 : n.kind === 'root' ? 0.65 : 0.4, fill: 'none', strokeLinecap: 'round' }));
      els.push(h('path', { key: 'rh' + i, className: 'root-hit', 'data-skill-id': n.id, d, stroke: 'transparent', strokeWidth: 9, fill: 'none', style: { cursor: 'pointer' }, onClick: e => { if (this.state.mode) return; e.stopPropagation(); this.setState({ menu: { id: n.id, x: n.px, y: n.py } }); } }));
    });
    briefs.forEach((b, i) => {
      const pj = PROJECTS[b.id], active = sel === b.id, dim = sel && !active, dir = b.left ? -1 : 1;
      const x = b.x + 12, y = b.y + 8, w = cardW - 24, chipH = 124 * cardScale;
      // Tall traces occupy the inner lanes, so shorter routes never cross them.
      const gap = drag.laneSpacing?.gap || 10, lane = cx + dir * (8 + b.row * gap), portX = b.left ? x + w : x;
      const knee = Math.max(y+4, Math.min(groundY-10,drag['k:'+b.id]?.y ?? y+Math.abs(lane-portX)));
      const branch = `M${lane},${groundY}V${knee}L${portX},${y}`;
      handle('k:'+b.id,lane,knee,p=>({y:Math.max(y+4,Math.min(groundY-10,snap(p.y)))}),pj.name+' bend');
      handle('laneSpacing',lane,groundY,p=>({gap:Math.max(8,Math.min(14,Math.round((Math.abs(p.x-cx)-8)/Math.max(1,b.row))))}),pj.name+' lane spacing');
      handle('c:'+b.id,portX,y,p=>({dx:Math.max(b.left?-22:-146,Math.min(b.left?146:22,snap(p.x-portX+(drag['c:'+b.id]?.dx||0)))),dy:Math.max(-b.y+24,Math.min(groundY-chipH-b.y-16,snap(p.y-y+(drag['c:'+b.id]?.dy||0))))}),pj.name+' attachment');
      els.push(h('path', { key: 'br' + i, className: 'project-branch', 'data-project-id':b.id, d: branch, stroke: active ? accent : ink, strokeWidth: active ? 3 : 2, fill: 'none', strokeLinejoin:'round', opacity: dim ? 0.2 : 1 }));
      const d = `M${x},${y}H${x+w}V${y+chipH}H${x}Z`;
      els.push(h('path', { key: 'chip' + i, className: 'project-chip', 'data-project-id':b.id, d, fill: active ? '#eff5f1' : '#fff', stroke: active ? accent : '#a4b5ad', strokeWidth: active ? 2 : 1, opacity: dim ? 0.35 : 1 }));
      const portRatio = Math.min(.5,Math.SQRT2*10/Math.hypot(lane-portX,knee-y));
      els.push(h('circle', {key:'port'+i,className:'project-port',cx:portX+(lane-portX)*portRatio,cy:y+(knee-y)*portRatio,r:4,fill:accent,opacity:dim?.25:1}));
      const toggle = e => { e.stopPropagation(); if (this._didDrag) { this._didDrag = false; return; } this.setState(s => ({ selected: s.selected === b.id ? null : b.id, menu: null })); };
      const textX = x + 22;
      const card = h('div', { className: 'project-card', 'data-project-id': b.id, role: 'button', tabIndex: 0, 'aria-pressed': active, 'aria-label': pj.name + ': ' + pj.result,
        onPointerDown: e => this.dragStart(e, 'c:' + b.id), onClick: toggle, onKeyDown: e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); toggle(e); } },
        style: { width: '284px', height: '100px', display: 'flex', flexDirection: 'column', justifyContent: 'center', gap: '4px', color: '#263d35', fontFamily: font, opacity: dim ? 0.35 : 1, cursor: 'grab', touchAction: 'none' } },
        h('div', { style: { display: 'flex', alignItems: 'baseline', gap: '7px', flexWrap: 'wrap', font: '700 18px/1.08 Arial,sans-serif' } }, pj.name, h('span', { style: { font: '400 16px Arial,sans-serif', color: '#617169' } }, pj.year)),
        h('div', { style: { font: '400 17.5px/1.08 Arial,sans-serif', color: '#4b5a53' } }, pj.desc),
        h('div', { style: { font: '700 18.5px/1.08 Arial,sans-serif', color: accent } }, pj.result)
      );
      els.push(h('foreignObject', { key: 'fo' + i, x: textX, y: y + 12 * cardScale, width: 284, height: 100, style: { overflow: 'visible' } }, h('div',{style:{transform:`scaleY(${cardScale})`,transformOrigin:'top left'}},card)));
    });
    bNodes.forEach((n, i) => {
      const opacity = dimB(n) ? 0.23 : 1, color = accent;
      const open = e => { if (this.state.mode) return; e.stopPropagation(); if (this._didDrag) { this._didDrag = false; return; } const keyboard = e.type === 'keydown'; this.setState({ menu: { id: n.id, x: n.px, y: n.py } }, () => { if (keyboard) requestAnimationFrame(() => document.querySelector('.branch-menu button')?.focus()); }); };
      const keyboard = e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); open(e); } };
      const moveNode = e => this.state.routeEdit ? this.routeStart(e,'n:'+n.id,p=>({dx:cx+snap(p.x-cx)-n.px+(drag['n:'+n.id]?.dx||0),dy:Math.max(groundY+10,Math.min(H-10,snap(p.y)))-n.py+(drag['n:'+n.id]?.dy||0)})) : this.dragStart(e,'n:'+n.id);
      els.push(h('circle', { key: 'bn' + i, className: 'skill-dot', 'data-skill-id': n.id, cx: n.px, cy: n.py, r: n.kind === 'root' ? 5 : n.kind === 'group' ? 3.6 : 2.6, fill: color, opacity, filter: printing ? 'none' : 'url(#nodeGlow)', onPointerDown: moveNode, onClick: open, style: { cursor: 'grab' } }));
      if (!printing && this.props.showSubskillLabels === false && n.gen >= 3) return;
      els.push(h('text', { ...baseT, key: 'bt' + i, className: 'skill-label', 'data-skill-id': n.id, x: n.px + 9, y: n.py + 5, fontSize: fs, fontWeight: n.kind === 'terminal' ? 400 : 700, fill: n.kind === 'terminal' ? '#34463d' : color, strokeWidth: 2, opacity, role: 'button', tabIndex: 0, 'aria-label': n.name + ': branch style', onKeyDown: keyboard, onClick: open, onPointerDown: e => this.dragStart(e, 'n:' + n.id), style: { cursor: 'grab' } }, n.lines.map((line, j) => h('tspan', { key: j, x: n.px + 9, dy: j ? lineH : 0 }, line))));
    });
    // ORIGINAL_EDITOR
    els.push(...handles);
    return h('svg', { id: 'treeSvg', viewBox: `0 0 ${W} ${H}`, width: '100%', height: '100%', preserveAspectRatio: 'xMidYMid meet', onClick: e => this.canvasClick(e), onPointerMove: e => this.dragMove(e), onPointerUp: () => this.dragEnd(), onPointerLeave: () => { if (!this._routeDrag) this.dragEnd(); }, onPointerCancel: () => this.dragEnd(), onLostPointerCapture: () => this.dragEnd(), style: { display: 'block', touchAction: 'none' } }, ...els);
  }
